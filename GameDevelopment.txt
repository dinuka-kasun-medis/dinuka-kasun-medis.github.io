<html>
    <body>
        <script src="js/three.js"></script>
        <script src="js/keyboard.js"></script>
        <script>
            var scene = new THREE.Scene();//create a new scene
            var camera=new THREE.PerspectiveCamera(75,window.innerWidth/window.innerHeight,0.1,1000);//camera angle eka, innerwidth ekei inner height eke ratio eka, near clip distance,far clip distance
            
            var renderer = new THREE.WebGLRenderer();//render kiynne camera krna ekkena wage..
            renderer.setSize(window.innerWidth,window.innerHeight);//
            document.body.appendChild(renderer.domElement);//renderrer ta body eke denawa wade keranna..
            
            var geometry = new THREE.BoxGeometry(1,1,1);
            var material = new THREE.MeshBasicMaterial( { color: 0xffff00 } );
            var material2 = new THREE.MeshBasicMaterial( { color: 0xff0000 } );
            var cube1 = new THREE.Mesh( geometry, material );
            var cube2 = new THREE.Mesh( geometry, material2 );
            
            var floor_geo = new THREE.BoxGeometry(25,0.1,20);
            var floor_material = new THREE.MeshBasicMaterial({color:0xff0aff});
            var floor = new THREE.Mesh(floor_geo,floor_material);
            			
			scene.add( cube1 );
            scene.add( cube2 );
            scene.add(floor);
            
            floor.position.y =-3;
            
            var state =0;

			camera.position.z = 15;
            camera.position.x = -10;
            camera.position.y =10;
            
            cube1.position.z =3;
            cube1.position.x =-8;
            
            cube2.position.z =3;
            cube2.position.x =8;
            
            
            camera.lookAt(floor.position);
            
            
            var clock = new THREE.Clock();
            clock.getDelta();
            var cubeVelocity=1;
            
            var animate= function(){
                
                requestAnimationFrame(animate);
                
                deltaTime = clock.getDelta();
                
                if(Key.isDown(Key.SPACE)){
                    state=1;
                }    
                
                if(state==1){
                    if(cube1.position.x>7){
                        state=2;
                    }
                    
                    cube1.position.x += cubeVelocity * deltaTime;
                    
                    cubeVelocity +=2*deltaTime;
                    //camera.position.x *=0.1;
                }
                if(state==2){
                    cube2.rotation.x +=0.1;
                    cube2.position.z +=0.1;
                }
                renderer.render(scene, camera);
            }

            animate();
//            renderer.render(scene, camera);
            
        </script>
    </body>
</html>
